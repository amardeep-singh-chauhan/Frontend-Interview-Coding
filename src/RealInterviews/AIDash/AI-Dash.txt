const arr = [2, 5, 8, 9, 10];
const condition = (num) => num % 2 === 0;

const filterMethod = (arr, con) => {
    const filteredArr = [];
    for (i = 0; i < arr.length; i++) {
        if (con(arr[i])) {
            filteredArr.push(arr[i]);
        }
    }

    return filteredArr;
}
